cmake_minimum_required(VERSION 3.20)
project(artifexian_worldgen_cpp VERSION 0.5.0 LANGUAGES CXX)
option(WORLDGEN_BUILD_TESTS "Build tests" ON)
option(WORLDGEN_ENABLE_SANITIZERS "Enable ASan/UBSan" OFF)
option(WORLDGEN_ENABLE_NATIVE "Enable -march=native" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(worldgen src/config.cpp src/grid.cpp src/topology.cpp src/noise.cpp src/progress.cpp src/field_io.cpp src/refinement.cpp src/output.cpp src/chemistry.cpp src/planetary_physics.cpp src/profile.cpp src/generator.cpp)
target_include_directories(worldgen PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(worldgen PRIVATE /W4 /permissive-)
else()
  target_compile_options(worldgen PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(WORLDGEN_ENABLE_NATIVE)
    target_compile_options(worldgen PRIVATE -march=native)
  endif()
  if(WORLDGEN_ENABLE_SANITIZERS)
    target_compile_options(worldgen PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(worldgen PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(worldgen_cli src/main.cpp)
target_link_libraries(worldgen_cli PRIVATE worldgen)
set_target_properties(worldgen_cli PROPERTIES OUTPUT_NAME worldgen)
if(WORLDGEN_BUILD_TESTS)
  enable_testing()
  add_executable(worldgen_tests tests/test_main.cpp tests/test_raster.cpp tests/test_topology.cpp tests/test_rng.cpp tests/test_config.cpp tests/test_noise.cpp tests/test_refinement.cpp tests/test_planetary_physics.cpp tests/test_profile.cpp tests/test_generator.cpp)
  target_link_libraries(worldgen_tests PRIVATE worldgen)
  add_test(NAME worldgen_tests COMMAND worldgen_tests)
endif()
